cmake_minimum_required(VERSION 3.20)
project(PyraUploadedDependencies LANGUAGES C CXX)

# Adapter for the flat, partly Windows-configured sources in Pyra_dep.zip.
# Copy sources before replacing platform configs; the upload stays untouched.
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
	message(FATAL_ERROR "This dependency adapter requires 64-bit Linux.")
endif()
set(PYRA_DEPENDENCY_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "Extracted Pyra_dep.zip directory")
option(PYRA_DEPENDENCY_ZLIB "Build uploaded zlib for TIFF Deflate/PixarLog" ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_STANDARD 99)
find_package(Threads REQUIRED)
include(TestBigEndian)
test_big_endian(PYRA_BIG_ENDIAN)
if(PYRA_BIG_ENDIAN)
	message(FATAL_ERROR "The uploaded-config adapter supports little-endian Linux only.")
endif()
foreach(folder jpegturbo3_1_3 libtiff47 openjp25 OpenJPH httplib)
	if(NOT IS_DIRECTORY "${PYRA_DEPENDENCY_SOURCE_DIR}/${folder}")
		message(FATAL_ERROR "Missing ${folder} in PYRA_DEPENDENCY_SOURCE_DIR")
	endif()
endforeach()

set(stage "${CMAKE_CURRENT_BINARY_DIR}/sources")
file(MAKE_DIRECTORY "${stage}")
foreach(folder jpegturbo3_1_3 libtiff47 openjp25)
	file(COPY "${PYRA_DEPENDENCY_SOURCE_DIR}/${folder}" DESTINATION "${stage}")
endforeach()
set(jpeg "${stage}/jpegturbo3_1_3")
set(tiff "${stage}/libtiff47")
set(opj "${stage}/openjp25")
file(READ "${jpeg}/jconfig.h" jpeg_config)
string(REPLACE "#define WITH_SIMD 1" "/* SIMD disabled: no assembler required. */" jpeg_config "${jpeg_config}")
file(WRITE "${jpeg}/jconfig.h" "${jpeg_config}")
configure_file(jconfigint.h.in "${jpeg}/jconfigint.h" @ONLY)

# Compile sample-dependent modules once for each supported precision.
set(jpeg_common jcapimin jcarith jchuff jcicc jcinit jclhuff jcmarker jcmaster
	jcomapi jcparam jcphuff jctrans jdapimin jdarith jdatadst jdatasrc jdhuff
	jdicc jdinput jdlhuff jdmarker jdmaster jdphuff jdtrans jerror jfdctflt
	jmemmgr jmemnobs jpeg_nbits jaricom)
set(jpeg_all jcapistd jccolor jcdiffct jclossls jcmainct jcprepct jcsample
	jdapistd jdcolor jddiffct jdlossls jdmainct jdpostct jdsample jutils)
set(jpeg_dct jccoefct jcdctmgr jdcoefct jddctmgr jdmerge jfdctfst jfdctint
	jidctflt jidctfst jidctint jidctred jquant1 jquant2)
set(jpeg_sources)
foreach(unit IN LISTS jpeg_common)
	list(APPEND jpeg_sources "${jpeg}/src/${unit}.c")
endforeach()
foreach(bits 8 12 16)
	set(units ${jpeg_all})
	if(NOT bits EQUAL 16)
		list(APPEND units ${jpeg_dct})
	endif()
	foreach(unit IN LISTS units)
		set(wrapper "${CMAKE_CURRENT_BINARY_DIR}/jpeg-wrappers/${unit}-${bits}.c")
		file(WRITE "${wrapper}" "#define BITS_IN_JSAMPLE ${bits}\n#include \"${jpeg}/src/${unit}.c\"\n")
		list(APPEND jpeg_sources "${wrapper}")
	endforeach()
endforeach()
add_library(jpeg STATIC ${jpeg_sources})
target_include_directories(jpeg PUBLIC "${jpeg}" "${jpeg}/src")
target_link_libraries(jpeg PUBLIC m)

set(SIZEOF_SIZE_T ${CMAKE_SIZEOF_VOID_P})
set(HOST_BIG_ENDIAN 0)
set(TIFF_MAX_DIR_COUNT 1048576)
foreach(bits 8 16 32 64)
	set(TIFF_INT${bits}_T int${bits}_t)
	set(TIFF_UINT${bits}_T uint${bits}_t)
endforeach()
set(TIFF_SSIZE_T ptrdiff_t)
foreach(feature HAVE_IEEEFP CCITT_SUPPORT JPEG_SUPPORT LOGLUV_SUPPORT LZW_SUPPORT
	NEXT_SUPPORT PACKBITS_SUPPORT THUNDER_SUPPORT SUBIFD_SUPPORT
	CHECK_JPEG_YCBCR_SUBSAMPLING HAVE_ASSERT_H HAVE_FCNTL_H HAVE_FSEEKO HAVE_MMAP
	HAVE_STRINGS_H HAVE_SYS_TYPES_H HAVE_UNISTD_H)
	set(${feature} 1)
endforeach()
if(PYRA_DEPENDENCY_ZLIB)
	if(NOT EXISTS "${PYRA_DEPENDENCY_SOURCE_DIR}/zlib132/zlib.h")
		message(FATAL_ERROR "The updated Pyra_dep.zip must contain zlib132, or explicitly disable PYRA_DEPENDENCY_ZLIB.")
	endif()
	set(zlib "${PYRA_DEPENDENCY_SOURCE_DIR}/zlib132")
	file(GLOB zlib_sources CONFIGURE_DEPENDS "${zlib}/*.c")
	add_library(z STATIC ${zlib_sources})
	target_include_directories(z PUBLIC "${zlib}")
	target_compile_definitions(z PRIVATE _POSIX_C_SOURCE=200809L HAVE_UNISTD_H)
	set(ZIP_SUPPORT 1)
	set(PIXARLOG_SUPPORT 1)
endif()
configure_file("${tiff}/tif_config.h.cmake.in" "${tiff}/tif_config.h" @ONLY)
configure_file("${tiff}/tiffconf.h.cmake.in" "${tiff}/tiffconf.h" @ONLY)
file(APPEND "${tiff}/tif_config.h" "\n#define HAVE_JPEGTURBO_DUAL_MODE_8_12 1\n")
file(GLOB tiff_sources CONFIGURE_DEPENDS "${tiff}/tif_*.c")
list(FILTER tiff_sources EXCLUDE REGEX "tif_win32\\.c$")
add_library(tiff STATIC ${tiff_sources})
target_include_directories(tiff PUBLIC "${tiff}")
target_compile_definitions(tiff PRIVATE _FILE_OFFSET_BITS=64)
target_link_libraries(tiff PUBLIC jpeg m)
if(PYRA_DEPENDENCY_ZLIB)
	target_link_libraries(tiff PUBLIC z)
endif()

configure_file(opj_config_private.h.in "${opj}/opj_config_private.h" @ONLY)
file(GLOB opj_sources CONFIGURE_DEPENDS "${opj}/*.c")
list(FILTER opj_sources EXCLUDE REGEX "/(bench_dwt|test_sparse_array|t1_generate_luts|t1_ht_generate_luts)\\.c$")
list(FILTER opj_sources EXCLUDE REGEX "_manager\\.c$")
add_library(openjp2 STATIC ${opj_sources})
target_compile_definitions(openjp2 PRIVATE MUTEX_pthread)
target_include_directories(openjp2 PUBLIC "${opj}")
target_link_libraries(openjp2 PUBLIC Threads::Threads m)

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(OJPH_BUILD_EXECUTABLES OFF CACHE BOOL "" FORCE)
set(OJPH_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(OJPH_ENABLE_TIFF_SUPPORT OFF CACHE BOOL "" FORCE)
add_subdirectory("${PYRA_DEPENDENCY_SOURCE_DIR}/OpenJPH" "${CMAKE_CURRENT_BINARY_DIR}/openjph")

enable_testing()
add_executable(pyra_dependency_test dependency_test.cpp)
target_compile_definitions(pyra_dependency_test PRIVATE PYRA_TEST_ZLIB=$<BOOL:${PYRA_DEPENDENCY_ZLIB}>)
target_link_libraries(pyra_dependency_test PRIVATE tiff openjp2 openjph)
target_include_directories(pyra_dependency_test PRIVATE "${PYRA_DEPENDENCY_SOURCE_DIR}/OpenJPH/src/core")
target_include_directories(pyra_dependency_test SYSTEM PRIVATE "${PYRA_DEPENDENCY_SOURCE_DIR}/httplib")
add_test(NAME pyra_uploaded_codecs COMMAND pyra_dependency_test "${CMAKE_CURRENT_BINARY_DIR}/test_out")

install(TARGETS jpeg tiff openjp2 ARCHIVE DESTINATION lib)
if(PYRA_DEPENDENCY_ZLIB)
	install(TARGETS z ARCHIVE DESTINATION lib)
	install(FILES "${zlib}/zlib.h" "${zlib}/zconf.h" DESTINATION include)
	install(FILES "${zlib}/zlib.h" DESTINATION share/pyra-dependencies RENAME zlib-license-header.h)
endif()
install(FILES "${jpeg}/src/jpeglib.h" "${jpeg}/src/jmorecfg.h" "${jpeg}/src/jerror.h"
	"${jpeg}/jconfig.h" "${tiff}/tiff.h" "${tiff}/tiffio.h" "${tiff}/tiffvers.h"
	"${tiff}/tiffconf.h" "${opj}/openjpeg.h" "${opj}/opj_config.h" "${opj}/opj_stdint.h" DESTINATION include)
install(FILES "${PYRA_DEPENDENCY_SOURCE_DIR}/httplib/httplib.h" DESTINATION include)
install(FILES "${jpeg}/LICENSE.md" DESTINATION share/pyra-dependencies RENAME libjpeg-turbo-LICENSE.md)
install(FILES "${jpeg}/README.ijg" DESTINATION share/pyra-dependencies OPTIONAL)
install(FILES "${tiff}/tif_version.c" "${opj}/openjpeg.c" DESTINATION share/pyra-dependencies)
install(FILES "${PYRA_DEPENDENCY_SOURCE_DIR}/OpenJPH/LICENSE" DESTINATION share/pyra-dependencies RENAME OpenJPH-LICENSE)
